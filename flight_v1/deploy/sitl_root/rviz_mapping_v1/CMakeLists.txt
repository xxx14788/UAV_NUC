cmake_minimum_required(VERSION 3.10)
project(realmap_viewer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp plan_env cv_bridge message_filters sensor_msgs nav_msgs geometry_msgs std_msgs)
find_package(PCL REQUIRED COMPONENTS common)
find_package(Eigen3 REQUIRED)
add_executable(map_only_node map_only_node.cpp)
target_include_directories(map_only_node PRIVATE ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
target_link_libraries(map_only_node ${catkin_LIBRARIES} ${PCL_LIBRARIES})
